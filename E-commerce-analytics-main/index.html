Source Code

HiveQL — ecommerce_analysis.hql 
CREATE DATABASE IF NOT EXISTS ecommerce_db;
USE ecommerce_db;

CREATE TABLE IF NOT EXISTS transactions_staging (
  order_id STRING, customer_id STRING, purchase_date STRING,
  product_id STRING, product_name STRING, category STRING,
  quantity INT, unit_price DOUBLE, discount_pct DOUBLE,
  total_amount DOUBLE, city STRING, payment_method STRING,
  customer_segment STRING
);

SELECT SUBSTR(purchase_date,1,7) AS month,
       ROUND(SUM(total_amount),2) AS total_sales,
       COUNT(*) AS transactions
FROM transactions_staging
GROUP BY SUBSTR(purchase_date,1,7)
ORDER BY month;

SELECT city, ROUND(SUM(total_amount),2) AS total_sales
FROM transactions_staging GROUP BY city
ORDER BY total_sales DESC LIMIT 5;

SELECT category, COUNT(*) AS transactions,
       ROUND(SUM(total_amount),2) AS total_sales
FROM transactions_staging GROUP BY category
ORDER BY total_sales DESC;
Pig Latin — clean_transactions.pig 
raw = LOAD '/ecommerce_project/raw/ecommerce_transactions.csv'
      USING PigStorage(',')
      AS (order_id:chararray, customer_id:chararray,
          purchase_date:chararray, product_id:chararray,
          product_name:chararray, category:chararray,
          quantity:int, unit_price:double,
          discount_pct:double, total_amount:double,
          city:chararray, payment_method:chararray,
          customer_segment:chararray);

valid = FILTER raw BY order_id IS NOT NULL
                    AND customer_id IS NOT NULL
                    AND product_id IS NOT NULL;

dedup = DISTINCT valid;
clean = FILTER dedup BY quantity IS NOT NULL AND quantity > 0;

STORE clean INTO '/ecommerce_project/cleaned'
USING PigStorage(',');
